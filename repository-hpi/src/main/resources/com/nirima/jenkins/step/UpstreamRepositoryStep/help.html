If selected, this defines Jenkins.Repository as the URL of the specified repository.
The recommended way of using this is to create a profile in the Jenkins settings.xml: 
        <profile>
            <id>jenkins-maven-upstream</id>
            <activation>
                <property>
                    <name>env.Jenkins.Repository</name>
                </property>
            </activation>
            <repositories>
                <repository>
                    <id>jenkins</id>
                    <url>${env.Jenkins.Repository}</url>
                    <releases>
                        <enabled>true</enabled>
                    </releases>
                    <snapshots>
                        <enabled>true</enabled>
                    </snapshots>
                </repository>
            </repositories>
            <pluginRepositories>
                <pluginRepository>
                    <id>jenkins</id>
                    <url>${env.Jenkins.Repository}</url>
                    <releases>
                        <enabled>true</enabled>
                    </releases>
                    <snapshots>
                        <enabled>true</enabled>
                    </snapshots>
                </pluginRepository>
            </pluginRepositories>
        </profile>

This profile is then automatically enabled.
